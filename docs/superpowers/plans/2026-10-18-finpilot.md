# FinPilot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build FinPilot, a multi-agent financial copilot (categorize → analyze → forecast → simulate → advise → verify) with measured accuracy, for Agent Arena 2026 Track 02.

**Architecture:** Deterministic pandas/numpy engines compute every number; LLM agents (via Mesh API) categorize unknown merchants, narrate findings, write advice, and answer chat. A Verifier agent recomputes every number an Advisor cites from raw transactions and rejects ungrounded advice. A LangGraph state machine wires a Coordinator route to the agents; a FastAPI + HTMX UI streams an agent trace and hosts the decision duel.

**Tech Stack:** Python 3.11+, FastAPI, SQLAlchemy 2 + SQLite, pandas, numpy, LangGraph, APScheduler, OpenAI SDK pointed at Mesh, Jinja2 + HTMX + Chart.js (CDN), pytest.

## Global Constraints

Copied from the spec (`docs/superpowers/specs/2026-10-18-finpilot-design.md`):

- All LLM calls go through Mesh API: `base_url="https://api.meshapi.ai/v1"`, key from `MESH_API_KEY`, via the OpenAI SDK, in the single module `finpilot/llm.py`. Default model `openai/gpt-4o-mini`, overridable with `FINPILOT_MODEL`.
- `requirements.txt` must list a web framework (`fastapi`) and an LLM client (`openai`).
- `.env` is gitignored; `.env.example` is committed. Never commit secrets.
- CI workflow lives at `.github/workflows/agent-arena-2026-checks.yml` (downloaded from the hackathon URL).
- No number in any recommendation originates from the LLM; engines compute, Verifier checks.
- LLM failure must degrade, never crash: Categorizer → rules + `uncategorized`; Advisor → templated text from evidence; Verifier still runs.
- Conventions: amounts are signed floats (negative = outflow); transaction DataFrame columns are `date` (datetime64), `description`, `amount`, then after categorization `merchant`, `category`, `category_source`, `confidence`.
- Cut line (drop in this order if short on time): behavior triggers, pace alerts, real-data check. Never the trace panel or the duel.
- Submission deadline Oct 30, 2026 11:00 PM IST.

```python
import json
import os
import time

from openai import OpenAI

from . import config


class LLMUnavailable(Exception):
    pass


class LLM:
    """All LLM traffic in FinPilot goes through this class, pointed at Mesh API."""

    def __init__(self, client=None, model=None):
        self.model = model or config.MODEL
        self._client = client

    @property
    def client(self):
        if self._client is None:
            key = os.getenv("MESH_API_KEY")
            if not key:
                raise LLMUnavailable("MESH_API_KEY not set")
            self._client = OpenAI(base_url=config.MESH_BASE_URL, api_key=key)
        return self._client

    def complete(self, system, user, json_mode=False):
        last = None
        for _ in range(2):
            try:
                kw = {"response_format": {"type": "json_object"}} if json_mode else {}
                r = self.client.chat.completions.create(
                    model=self.model, temperature=0,
                    messages=[{"role": "system", "content": system},
                              {"role": "user", "content": user}], **kw)
                return r.choices[0].message.content or ""
            except LLMUnavailable:
                raise
            except Exception as e:  # network, rate limit, bad model
                last = e
                time.sleep(1)
        raise LLMUnavailable(str(last))

    def json(self, system, user):
        txt = self.complete(system, user, json_mode=True)
        try:
            return json.loads(txt)
        except json.JSONDecodeError as e:
            raise LLMUnavailable(f"bad json from model: {e}")


_llm = None


def get_llm():
    global _llm
    if _llm is None:
        _llm = LLM()
    return _llm


def set_llm(obj):
    global _llm
    _llm = obj
```

```python
"""Synthetic personas with known ground truth for evaluation and demo."""
from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd

# (display name, category, kind, params). Rows marked NOT-IN-RULES are deliberately
# absent from the keyword rules so the LLM fallback has real work to do.
STREAMS = [
    ("SUNRISE APARTMENTS RENT", "rent", "monthly", {"day": 5, "amt": 18000}),
    ("BESCOM ELECTRICITY", "utilities", "monthly", {"day": 12, "amt": 1500, "sd": 250}),
    ("AIRTEL BROADBAND", "utilities", "monthly", {"day": 15, "amt": 999}),

    ("HOTSTAR", "subscriptions", "monthly", {"day": 8, "amt": 299}),
    ("SPOTIFY", "subscriptions", "monthly", {"day": 20, "amt": 119}),
    ("CULT FIT GYM", "subscriptions", "monthly", {"day": 10, "amt": 1500}),
    ("BIGBASKET", "groceries", "poisson", {"per_month": 5, "amt": 1600, "sd": 400}),
    ("DMART", "groceries", "poisson", {"per_month": 3, "amt": 1100, "sd": 300}),
    ("SWIGGY", "food_delivery", "poisson", {"per_month": 4, "amt": 380, "sd": 90}),

    ("APOLLO PHARMACY", "health", "poisson", {"per_month": 1, "amt": 650, "sd": 250}),
    ("DR SHAH CLINIC", "health", "poisson", {"per_month": 0.4, "amt": 1200, "sd": 300}),  # NOT-IN-RULES
    ("REDBUS", "travel", "poisson", {"per_month": 0.5, "amt": 900, "sd": 200}),  # NOT-IN-RULES
]

PERSONAS = {
    "priya": {"seed": 1, "salary": 62000, "gym_used": False},
    "arjun": {"seed": 2, "salary": 58000, "gym_used": False},
    "meera": {"seed": 3, "salary": 70000, "gym_used": True},
}


@dataclass
class Truth:
    categories: pd.Series
    anomaly_ids: set
    duplicate_ids: set
    unused_subs: set
    spike: tuple
    payday_trigger_category: str


@dataclass
class Persona:
    df: pd.DataFrame
    truth: Truth
    opening_balance: float


def generate(seed=0, months=12, end="2026-09", salary=62000, gym_used=False) -> Persona:
    rng = np.random.default_rng(seed)
    periods = pd.period_range(end=end, periods=months, freq="M")
    rows = []

    def desc(name):
        ref = int(rng.integers(100000, 999999))
        return str(rng.choice([f"UPI-{name}-{ref}", f"POS {name} BANGALORE", name, f"{name} PAYMENT {ref}"]))

    def add(date, name, cat, amt, tag=None):
        rows.append(dict(date=date, description=desc(name), amount=round(float(amt), 2),
                         category=cat, tag=tag))

    spike_p, dup_p, big_p = periods[-3], periods[-4], periods[-5]
    streams = list(STREAMS)
    if gym_used:
        streams.append(("DECATHLON", "shopping", "poisson", {"per_month": 1, "amt": 2500, "sd": 500}))

    for p in periods:
        dim = p.days_in_month

        def d(day, p=p, dim=dim):
            return p.to_timestamp() + pd.Timedelta(days=min(day, dim) - 1)

        add(d(1), "ACME CORP SALARY", "income", salary)
        for name, cat, kind, prm in streams:
            if kind == "monthly":
                add(d(prm["day"]), name, cat, -max(1.0, rng.normal(prm["amt"], prm.get("sd", 0))))
            else:
                mult = 3.0 if (p == spike_p and cat == "food_delivery") else 1.0
                for day in rng.integers(1, dim + 1, rng.poisson(prm["per_month"] * mult)):
                    add(d(int(day)), name, cat, -max(50.0, rng.normal(prm["amt"], prm["sd"])))
        for day in rng.integers(1, 4, 3):  # payday behavior trigger
            add(d(int(day)), "SWIGGY", "food_delivery", -max(50.0, rng.normal(380, 90)), "payday")
        if p == big_p:
            add(d(17), "GLOBALTECH TRADING LTD", "shopping", -24000, "big")
        if p == dup_p:
            add(d(21), "AMAZON", "shopping", -2199.0, "dup_orig")
            add(d(22), "AMAZON", "shopping", -2199.0, "dup")

    df = pd.DataFrame(rows).sort_values("date", kind="stable").reset_index(drop=True)
    unused = {"HOTSTAR"} if gym_used else {"HOTSTAR", "CULT FIT GYM"}
    truth = Truth(
        categories=df["category"].copy(),
        anomaly_ids=set(df.index[df.tag == "big"]),
        duplicate_ids=set(df.index[df.tag == "dup"]),
        unused_subs=unused,
        spike=(str(spike_p), "food_delivery"),
        payday_trigger_category="food_delivery",
    )
    return Persona(df[["date", "description", "amount"]], truth, 40000.0)


def load_persona(name) -> Persona:
    return generate(**PERSONAS[name])
```

```python
import re

STOP = {"UPI", "POS", "NEFT", "IMPS", "ACH", "PAYMENT", "BANGALORE", "BLR", "MUMBAI", "DELHI", "PUNE"}

RULES = {  # order matters: first match wins
    "rent": ["RENT"],
    "utilities": ["ELECTRICITY", "BROADBAND", "WATER", "GAS", "BESCOM", "AIRTEL", "JIO FIBER"],
    "subscriptions": ["NETFLIX", "SPOTIFY", "HOTSTAR", "PRIME VIDEO", "CULT FIT", "ADOBE", "YOUTUBE"],
    "groceries": ["BIGBASKET", "DMART", "GROFERS", "SUPERMARKET"],
    "food_delivery": ["SWIGGY", "ZOMATO"],
    "dining": ["RESTAURANT", "CAFE"],
    "transport": ["UBER", "OLA", "METRO", "RAPIDO"],
    "shopping": ["AMAZON", "FLIPKART", "MYNTRA"],
    "health": ["PHARMACY", "HOSPITAL"],
}
_COMPILED = [(c, re.compile(r"\b(" + "|".join(re.escape(k) for k in ks) + r")\b"))
             for c, ks in RULES.items()]


def normalize_merchant(desc: str) -> str:
    toks = re.findall(r"[A-Za-z]+", str(desc).upper())
    return " ".join(t for t in toks if t not in STOP and len(t) > 1)


def rule_category(merchant: str, amount: float):
    if amount > 0:
        return "income"
    for cat, rx in _COMPILED:
        if rx.search(merchant):
            return cat
    return None
```

```python
    if not unknown or llm is None:

    unknown = sorted(unknown)
    for i in range(0, len(unknown), 40):
        batch = unknown[i:i + 40]
        try:
            res = llm.json(SYSTEM, json.dumps(batch))
        except (LLMUnavailable, ValueError):
            continue
        for m in batch:
            c = res.get(m)
            if c is not None:
                mapping[m] = c if c in CATEGORIES else "other"
    return mapping


def categorize_df(df, llm=None, overrides=None):
    out = df.copy()
    out["merchant"] = out["description"].map(normalize_merchant)
    overrides = overrides or {}
    cats, srcs, confs, unknown = [], [], [], set()
    for m, amt in zip(out["merchant"], out["amount"]):
        if m in overrides:
            cats.append(overrides[m]); srcs.append("user"); confs.append(1.0)
            continue
        c = rule_category(m, amt)
        if c:
            cats.append(c); srcs.append("rule"); confs.append(0.95)
        else:
            cats.append(None); srcs.append(None); confs.append(0.0)
            unknown.add(m)
    mapping = _llm_map(unknown, llm)
    for i, m in enumerate(out["merchant"]):
        if cats[i] is None:
            if m in mapping:
                cats[i], srcs[i], confs[i] = mapping[m], "llm", 0.75
            else:
                cats[i], srcs[i] = "uncategorized", "none"
    out["category"], out["category_source"], out["confidence"] = cats, srcs, confs
    return out
```

```python
from __future__ import annotations

import numpy as np
import pandas as pd

GROUPS = {
    "video": {"NETFLIX", "HOTSTAR", "PRIME VIDEO", "DISNEY"},
    "music": {"SPOTIFY", "GAANA", "YOUTUBE MUSIC"},
    "fitness": {"CULT FIT GYM", "GOLDS GYM", "ANYTIME FITNESS"},
}
USAGE_KEYWORDS = {"fitness": ["DECATHLON", "PROTEIN", "SPORTS"]}


def _mad(s: pd.Series, med: float) -> float:
    mad = float((s - med).abs().median())
    return mad if mad > 0 else max(0.05 * abs(med), 1.0)


def monthly_aggregates(df):
    spend = df[df["amount"] < 0].copy()
    spend["month"] = spend["date"].dt.to_period("M").astype(str)
    spend["spend"] = -spend["amount"]
    return spend.pivot_table(index="month", columns="category", values="spend",
                             aggfunc="sum", fill_value=0.0)


def detect_recurring(df, min_n=3):
    out = []
    for m, g in df.groupby("merchant"):
        if len(g) < min_n:
            continue
        g = g.sort_values("date")
        gaps = g["date"].diff().dt.days.dropna()
        if not (26 <= gaps.median() <= 35) or gaps.std(ddof=0) > 5:
            continue
        amt = g["amount"].abs()
        cv = float(amt.std(ddof=0) / amt.mean()) if amt.mean() else 0.0
        if cv > 0.25:
            continue
        out.append({"merchant": m, "category": g["category"].mode().iloc[0],
                    "amount": float(g["amount"].mean()), "day": int(g["date"].dt.day.median()),
                    "n": len(g), "cv": round(cv, 3), "last_date": g["date"].max()})
    return out


def detect_subscriptions(df):
    out = []
    for r in detect_recurring(df):
        if r["category"] == "subscriptions" and r["cv"] <= 0.1 and r["amount"] < 0:
            mc = round(abs(r["amount"]), 2)
            out.append({"id": f"sub:{r['merchant']}", "type": "subscription",
                        "merchant": r["merchant"], "monthly_cost": mc, "annual_cost": round(mc * 12, 2)})
    return out


def _unused(sub, reason, detail):
    return {"id": f"unused:{sub['merchant']}", "type": "unused_subscription",
            "merchant": sub["merchant"], "monthly_cost": sub["monthly_cost"],
            "annual_cost": sub["annual_cost"], "reason": reason, "detail": detail}


def unused_subscriptions(df, subs):
    """Candidates only: overlap within a service group, or a group with a usage
    signal (e.g. fitness) where no related spending exists. Precision is measured in eval."""
    active = {s["merchant"]: s for s in subs}
    merchants = set(df["merchant"])
    out, flagged = [], set()
    for grp, members in GROUPS.items():
        present = [m for m in members if m in active]
        if len(present) >= 2:
            keep = max(present, key=lambda m: active[m]["monthly_cost"])
            for m in present:
                if m != keep:
                    out.append(_unused(active[m], "overlap", f"overlaps with {keep} ({grp})"))
                    flagged.add(m)
        kws = USAGE_KEYWORDS.get(grp)
        if kws and not any(k in mm for mm in merchants for k in kws):
            for m in present:
                if m not in flagged:
                    out.append(_unused(active[m], "no_usage_signal", f"no related {grp} spending found"))
                    flagged.add(m)
    return out


def category_spikes(df, z=3.5, min_pct=0.5):
    agg = monthly_aggregates(df)
    out = []
    for cat in agg.columns:
        s = agg[cat]
        if len(s) < 4 or cat in ("rent", "income"):
            continue
        med = float(s.median())
        mad = _mad(s, med)
        for month, v in s.items():
            rz = 0.6745 * (v - med) / mad
            if rz > z and v > med * (1 + min_pct):
                out.append({"id": f"spike:{cat}:{month}", "type": "category_spike",
                            "category": cat, "month": month, "value": round(float(v), 2),
                            "baseline": round(med, 2), "pct_change": round((v / med - 1) * 100, 1)})
    return out


def txn_outliers(df, exclude_merchants=frozenset(), k=6.0, min_ratio=3.0):
    spend = df[(df["amount"] < 0) & (~df["merchant"].isin(exclude_merchants))].copy()
    spend["a"] = -spend["amount"]
    gmed = float(spend["a"].median())
    gmad = _mad(spend["a"], gmed)
    out = []
    for cat, g in spend.groupby("category"):
        med, mad = (float(g["a"].median()), _mad(g["a"], float(g["a"].median()))) if len(g) >= 8 else (gmed, gmad)
        for idx, row in g.iterrows():
            if row["a"] > med + k * 1.4826 * mad and row["a"] >= min_ratio * med:
                out.append({"id": f"outlier:{idx}", "type": "large_charge", "txn_id": int(idx),
                            "merchant": row["merchant"], "amount": round(float(row["a"]), 2),
                            "date": str(row["date"].date())})
    return out


def duplicate_charges(df, hours=48):
    spend = df[df["amount"] < 0].sort_values("date")
    out = []
    for (m, a), g in spend.groupby(["merchant", "amount"]):
        if len(g) < 2:
            continue
        dates, idxs = g["date"].tolist(), g.index.tolist()
        for i in range(1, len(g)):
            if dates[i] - dates[i - 1] <= pd.Timedelta(hours=hours):
                out.append({"id": f"dup:{idxs[i]}", "type": "duplicate_charge", "txn_id": int(idxs[i]),
                            "merchant": m, "amount": round(abs(float(a)), 2), "date": str(dates[i].date())})
    return out


def behavior_triggers(df, exclude_merchants=frozenset(), window_days=3, min_ratio=1.8, min_hits=6):
    paydays = np.sort(df.loc[df["category"] == "income", "date"].dt.normalize().unique())
    if len(paydays) < 3:
        return []
    spend = df[(df["amount"] < 0) & (~df["merchant"].isin(exclude_merchants))]
    out = []
    for cat, g in spend.groupby("category"):
        if len(g) < 20:
            continue
        d = g["date"].dt.normalize().to_numpy()
        pos = np.searchsorted(paydays, d, side="right") - 1
        ok = pos >= 0
        offset = (d[ok] - paydays[pos[ok]]).astype("timedelta64[D]").astype(int)
        hits = int((offset < window_days).sum())
        share = hits / len(g)
        ratio = share / (window_days / 30.4)
        if hits >= min_hits and ratio >= min_ratio:
            out.append({"id": f"trigger:{cat}:payday", "type": "behavior_trigger", "category": cat,
                        "ratio": round(ratio, 2), "share_pct": round(share * 100, 1),
                        "window_days": window_days})
    return out


def analyze_all(df):
    rec = {r["merchant"] for r in detect_recurring(df)}
    subs = detect_subscriptions(df)
    return (subs + unused_subscriptions(df, subs) + category_spikes(df)
            + txn_outliers(df, rec) + duplicate_charges(df) + behavior_triggers(df, rec))
```

```python
from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd

from .analyze import detect_recurring


@dataclass
class Model:
    recurring: list
    cats: list
    daily: np.ndarray       # days x cats, positive discretionary spend
    weekdays: np.ndarray    # weekday of each row in daily
    last_date: pd.Timestamp


def current_balance(df, opening):
    return float(opening + df["amount"].sum())


def fit(df, window_days=90) -> Model:
    rec = detect_recurring(df)
    rec_m = {r["merchant"] for r in rec}
    end = df["date"].max().normalize()
    idx = pd.date_range(end - pd.Timedelta(days=window_days - 1), end)
    disc = df[(~df["merchant"].isin(rec_m)) & (df["amount"] < 0) & (df["date"] >= idx[0])]
    cats = sorted(disc["category"].unique())
    if cats:
        mat = (disc.assign(d=disc["date"].dt.normalize(), s=-disc["amount"])
               .pivot_table(index="d", columns="category", values="s", aggfunc="sum")
               .reindex(idx).fillna(0.0).reindex(columns=cats, fill_value=0.0))
        daily = mat.to_numpy()
    else:
        daily = np.zeros((len(idx), 0))
    return Model(rec, cats, daily, np.asarray(idx.weekday), end)


def recurring_flows(model, dates):
    flows = np.zeros(len(dates))
    for r in model.recurring:
        for i, d in enumerate(dates):
            if d.day == min(r["day"], d.days_in_month):
                flows[i] += r["amount"]
    return flows


@dataclass
class Result:
    dates: pd.DatetimeIndex
    paths: np.ndarray

    @property
    def p10(self): return np.percentile(self.paths, 10, axis=0)
    @property
    def p50(self): return np.percentile(self.paths, 50, axis=0)
    @property
    def p90(self): return np.percentile(self.paths, 90, axis=0)
    @property
    def p_overdraft(self): return float((self.paths.min(axis=1) < 0).mean())
    @property
    def floor_p50(self): return float(np.median(self.paths.min(axis=1)))
    @property
    def end_p10(self): return float(self.p10[-1])
    @property
    def end_p50(self): return float(self.p50[-1])
    @property
    def end_p90(self): return float(self.p90[-1])

    @property
    def risk_day(self):
        frac = (self.paths < 0).mean(axis=0)
        hit = np.where(frac > 0.10)[0]
        return str(self.dates[hit[0]].date()) if len(hit) else None

    def summary(self):
        return {"p_overdraft": round(self.p_overdraft, 3), "floor_p50": round(self.floor_p50, 2),
                "end_p10": round(self.end_p10, 2), "end_p50": round(self.end_p50, 2),
                "end_p90": round(self.end_p90, 2), "risk_day": self.risk_day}


def monte_carlo(model, balance, horizon=60, n=1000, seed=0, extra_flows=(), cuts=None) -> Result:
    rng = np.random.default_rng(seed)
    dates = pd.date_range(model.last_date + pd.Timedelta(days=1), periods=horizon)
    fixed = recurring_flows(model, dates)
    for d, a in extra_flows:
        d = pd.Timestamp(d).normalize()
        if d in dates:
            fixed[dates.get_loc(d)] += a
    cuts = cuts or {}
    scale = np.array([1.0 - cuts.get(c, 0.0) for c in model.cats])
    by_wd = {w: np.where(model.weekdays == w)[0] for w in range(7)}
    rows = np.empty((n, horizon), dtype=int)
    for j, d in enumerate(dates):  # weekday-matched bootstrap of observed days
        rows[:, j] = rng.choice(by_wd[d.weekday()], size=n)
    disc = (model.daily[rows] * scale).sum(axis=2) if model.cats else np.zeros((n, horizon))
    paths = balance + np.cumsum(fixed[None, :] - disc, axis=1)
    return Result(dates, paths)


def goal_status(model, balance, target, deadline, n=1000, seed=0, max_horizon=365):
    deadline = pd.Timestamp(deadline).normalize()
    horizon = max(1, (deadline - model.last_date).days)
    r = monte_carlo(model, balance, min(horizon, max_horizon), n, seed)
    p_met = float((r.paths[:, -1] >= target).mean())
    far = monte_carlo(model, balance, max_horizon, n, seed)
    hit = np.where(far.p50 >= target)[0]
    return {"p_met": round(p_met, 3), "shortfall_p50": round(float(target - r.end_p50), 2),
            "eta": str(far.dates[hit[0]].date()) if len(hit) else None, "horizon": int(horizon)}
```

```python
from __future__ import annotations

import pandas as pd

from .forecast import monte_carlo

STEPS = (0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.60)


def purchase_flows(amount, on, financed_months=0):
    on = pd.Timestamp(on).normalize()
    if not financed_months:
        return [(on, -float(amount))]
    part = float(amount) / financed_months
    return [(on + pd.DateOffset(months=k), -part) for k in range(financed_months)]


def simulate_purchase(model, balance, amount, on, horizon=60, n=1000, seed=0, financed_months=0):
    flows = purchase_flows(amount, on, financed_months)
    base = monte_carlo(model, balance, horizon, n, seed)
    alt = monte_carlo(model, balance, horizon, n, seed, extra_flows=flows)
    b, a = base.summary(), alt.summary()
    return {"amount": float(amount), "date": str(pd.Timestamp(on).date()),
            "baseline": b, "with_purchase": a,
            "delta_p_overdraft": round(a["p_overdraft"] - b["p_overdraft"], 3),
            "delta_end_p50": round(a["end_p50"] - b["end_p50"], 2)}


def best_date(model, balance, amount, candidates, horizon=60, n=1000, seed=0):
    out = []
    for c in candidates:
        r = monte_carlo(model, balance, horizon, n, seed, extra_flows=purchase_flows(amount, c))
        out.append({"date": str(pd.Timestamp(c).date()), **r.summary()})
    return sorted(out, key=lambda x: (x["p_overdraft"], -x["floor_p50"]))


def cheapest_fix(model, balance, extra_flows, target_p=0.10, horizon=60, n=1000, seed=0):
    """Smallest discretionary cut (single top category, or uniform) that brings
    P(overdraft) under target_p. Cheapest = lowest monthly saving required."""
    base = monte_carlo(model, balance, horizon, n, seed, extra_flows=extra_flows)
    if base.p_overdraft <= target_p:
        return {"category": None, "cut_pct": 0, "p_overdraft": round(base.p_overdraft, 3),
                "est_monthly_saving": 0.0}
    if not model.cats:
        return None
    mean_daily = dict(zip(model.cats, model.daily.mean(axis=0)))
    top = sorted(mean_daily, key=mean_daily.get, reverse=True)[:3]
    options = [(c, [c]) for c in top] + [("all", list(model.cats))]
    best = None
    for name, cats in options:
        for s in STEPS:
            r = monte_carlo(model, balance, horizon, n, seed, extra_flows=extra_flows,
                            cuts={c: s for c in cats})
            if r.p_overdraft <= target_p:
                saving = float(sum(mean_daily[c] for c in cats) * 30 * s)
                cand = {"category": name, "cut_pct": int(round(s * 100)),
                        "p_overdraft": round(r.p_overdraft, 3), "est_monthly_saving": round(saving, 2)}
                if best is None or saving < best["est_monthly_saving"]:
                    best = cand
                break
    return best


def _bands(res):
    return {"dates": [str(d.date()) for d in res.dates],
            "p10": [round(float(x), 2) for x in res.p10],
            "p50": [round(float(x), 2) for x in res.p50],
            "p90": [round(float(x), 2) for x in res.p90]}


def duel(model, balance, a, b, horizon=90, n=1000, seed=0):
    ra = monte_carlo(model, balance, horizon, n, seed, extra_flows=a["flows"], cuts=a.get("cuts"))
    rb = monte_carlo(model, balance, horizon, n, seed, extra_flows=b["flows"], cuts=b.get("cuts"))
    sa, sb = ra.summary(), rb.summary()
    if abs(sa["p_overdraft"] - sb["p_overdraft"]) > 0.02:
        winner, basis = ("a" if sa["p_overdraft"] < sb["p_overdraft"] else "b"), "overdraft risk"
    elif sa["end_p50"] != sb["end_p50"]:
        winner, basis = ("a" if sa["end_p50"] > sb["end_p50"] else "b"), "end balance"
    else:
        winner, basis = "tie", "nothing"
    if winner == "tie":
        reason = "Both options end in the same place."
    else:
        w, l = (sa, sb) if winner == "a" else (sb, sa)
        wn = a["name"] if winner == "a" else b["name"]
        reason = (f"{wn} wins on {basis}: overdraft risk {w['p_overdraft']*100:.0f}% vs "
                  f"{l['p_overdraft']*100:.0f}%, median balance in {horizon} days "
                  f"₹{w['end_p50']:,.0f} vs ₹{l['end_p50']:,.0f}.")
    return {"a": {"name": a["name"], "summary": sa, "bands": _bands(ra)},
            "b": {"name": b["name"], "summary": sb, "bands": _bands(rb)},
            "winner": winner, "reason": reason, "horizon": horizon}
```

```python
from __future__ import annotations

import re

import pandas as pd

ALLOWED_CONSTANTS = {12, 24, 30, 60, 90, 100}


def _month(df):
    return df["date"].dt.to_period("M").astype(str)


def recompute(claim, df):
    k = claim["kind"]
    if k in ("monthly_cost", "annual_cost"):
        g = df[(df["merchant"] == claim["merchant"]) & (df["amount"] < 0)].sort_values("date").tail(3)
        if g.empty:
            return None
        v = float(abs(g["amount"].mean()))
        return v * 12 if k == "annual_cost" else v
    if k == "month_total":
        m = df[(df["category"] == claim["category"]) & (df["amount"] < 0) & (_month(df) == claim["month"])]
        return None if m.empty else float(-m["amount"].sum())
    if k == "pct_change":
        s = df[(df["category"] == claim["category"]) & (df["amount"] < 0)]
        tot = (-s["amount"]).groupby(_month(s)).sum()
        if claim["month"] not in tot.index or len(tot) < 3:
            return None
        return float((tot[claim["month"]] / tot.median() - 1) * 100)
    if k == "txn_amount":
        try:
            return float(abs(df.loc[int(claim["txn_id"]), "amount"]))
        except (KeyError, ValueError):
            return None
    return None


def check_claim(claim, df):
    actual = recompute(claim, df)
    if actual is None:
        return {"ok": False, "actual": None}
    tol = 3.0 if claim["kind"] == "pct_change" else max(1.0, 0.02 * abs(actual))
    return {"ok": abs(float(claim["value"]) - actual) <= tol, "actual": actual}


def _flatten_numbers(obj):
    if isinstance(obj, bool):
        return
    if isinstance(obj, (int, float)):
        yield float(obj)
    elif isinstance(obj, dict):
        for v in obj.values():
            yield from _flatten_numbers(v)
    elif isinstance(obj, (list, tuple)):
        for v in obj:
            yield from _flatten_numbers(v)


def ungrounded_numbers(text, claims, evidence):
    allowed = list(_flatten_numbers([c.get("value") for c in claims])) + list(_flatten_numbers(evidence))
    allowed += [a * 100 for a in allowed if 0 < abs(a) <= 1]  # probabilities quoted as percents
    text = re.sub(r"\d{4}-\d{2}(-\d{2})?", " ", text)  # ISO dates and months are not claims
    out = []
    for raw in re.findall(r"\d[\d,]*\.?\d*", text):
        raw = raw.replace(",", "").rstrip(".")
        if not raw:
            continue
        n = float(raw)
        if n < 10 or 1990 <= n <= 2100 or n in ALLOWED_CONSTANTS:
            continue
        if not any(abs(n - a) <= max(1.0, 0.01 * abs(a)) for a in allowed):
            out.append(n)
    return out
```

```python
                f"{e['window_days']} days after payday ({e['ratio']:.1f}x the normal rate). Delay those orders a few days.")
    if t == "forecast_risk":
        return (f"At the current pace there is a {e['p_overdraft']*100:.0f}% chance your balance goes negative "
                f"within {e['horizon']} days, first around {e['risk_day']}. Hold discretionary spending until then.")
    if t == "goal_behind":
        return f"You have a {e['p_met']*100:.0f}% chance of meeting '{e['name']}' by its deadline. Increase contributions or move the date."
    return rec["action"]


SYSTEM = ("You are a financial advisor. For each candidate write at most two sentences of specific advice "
          "using ONLY numbers present in its evidence or claims. Write full numbers (₹1500, not 1.5k). "
          "Return JSON {\"recommendations\":[{\"cid\":str,\"text\":str,\"claims\":[...]}]} where claims "
          "copies the candidate's claim objects, with 'value' set to the number you state in the text.")


def advise(candidates, llm=None, model_name=""):

    if llm is not None and candidates:

            by_cid = {r["cid"]: r for r in res.get("recommendations", []) if "cid" in r and "text" in r}
        except (LLMUnavailable, ValueError, AttributeError, TypeError):
            by_cid = {}
    recs = []
    for c in candidates:

                 model=model_name if c["cid"] in by_cid else "template")
        if c["cid"] in by_cid:
            r["text"] = str(by_cid[c["cid"]]["text"])
            llm_claims = by_cid[c["cid"]].get("claims")
            if isinstance(llm_claims, list) and llm_claims:
                r["claims"] = llm_claims
        else:
            r["text"] = template_text(r)
        recs.append(r)
    return recs
```

```python
from __future__ import annotations

import copy
import json

from ..engines.verify import check_claim, ungrounded_numbers
from ..llm import LLMUnavailable
from .advisor import template_text

REWRITE_SYSTEM = ("Rewrite the advice using ONLY the corrected claim values and the evidence. Keep it to "
                  "two sentences. Return JSON {\"text\": str}.")


def audit(rec, df):
    issues = []
    for c in rec["claims"]:
        r = check_claim(c, df)
        if r["actual"] is None:
            issues.append({"type": "unverifiable", "claim": c})
        elif not r["ok"]:
            issues.append({"type": "mismatch", "claim": c, "actual": round(r["actual"], 2)})
    for n in ungrounded_numbers(rec["text"], rec["claims"], rec["evidence"]):
        issues.append({"type": "ungrounded_number", "value": n})
    return issues


def _describe(issues):
    return "; ".join(
        f"{i['type']}: claimed {i['claim']['value']} vs actual {i['actual']}" if i["type"] == "mismatch"
        else f"{i['type']}: {i.get('claim') or i.get('value')}" for i in issues)


def verify_all(recs, df, llm=None, emit=None):
    emit = emit or (lambda kind, summary, data=None: None)
    kept, dropped = [], []
    for rec in recs:
        issues = audit(rec, df)
        if not issues:
            rec.update(verified=True, verifier_notes="all claims recomputed from transactions")
            emit("ok", f"verified: {rec['action']}", {"rec": rec["id"]})
            kept.append(rec)
            continue
        emit("caught", f"{rec['action']}: {_describe(issues)}", {"rec": rec["id"], "issues": issues})
        if any(i["type"] == "unverifiable" for i in issues):
            rec.update(verified=False, verifier_notes="dropped: " + _describe(issues))
            emit("dropped", f"dropped: {rec['action']} (claim cannot be recomputed)", {"rec": rec["id"]})
            dropped.append(rec)
            continue
        notes = "mismatch caught: " + _describe(issues)
        fixed = copy.deepcopy(rec["claims"])
        for i in issues:
            if i["type"] == "mismatch":
                for c in fixed:
                    if c == i["claim"]:
                        c["value"] = i["actual"]
        rewritten = None
        if llm is not None:
            try:
                rewritten = llm.json(REWRITE_SYSTEM, json.dumps(
                    {"evidence": rec["evidence"], "claims": fixed, "bad_text": rec["text"],
                     "issues": [str(i) for i in issues]}, default=str)).get("text")
            except (LLMUnavailable, ValueError, AttributeError):
                rewritten = None
        attempts = ([(rewritten, "LLM rewrite")] if rewritten else []) + [(None, "template")]
        for candidate_text, how in attempts:
            rec["claims"] = fixed
            rec["text"] = candidate_text if candidate_text else template_text(rec)
            if not audit(rec, df):
                rec.update(verified=True, verifier_notes=f"{notes}; corrected via {how}")
                emit("corrected", f"corrected via {how}: {rec['action']}", {"rec": rec["id"]})
                kept.append(rec)
                break
        else:
            rec.update(verified=False, verifier_notes=notes + "; could not be grounded")
            emit("dropped", f"dropped: {rec['action']}", {"rec": rec["id"]})
            dropped.append(rec)
    return kept, dropped
```

```python
import threading
import time
from collections import defaultdict


class Trace:
    def __init__(self):
        self._ev = defaultdict(list)
        self._done = set()
        self._lock = threading.Lock()

    def emit(self, run_id, agent, kind, summary, data=None):
        with self._lock:
            ev = {"seq": len(self._ev[run_id]), "ts": time.time(), "agent": agent,
                  "kind": kind, "summary": summary, "data": data or {}}
            self._ev[run_id].append(ev)
        return ev

    def since(self, run_id, n=0):
        with self._lock:
            return list(self._ev[run_id][n:])

    def finish(self, run_id):
        self.emit(run_id, "coordinator", "done", "run complete")
        with self._lock:
            self._done.add(run_id)

    def is_done(self, run_id):
        return run_id in self._done


trace = Trace()
```

```python
def inr(x):

def narrate(llm, system, payload, fallback):
    if llm is None:
        return fallback
    try:
        return llm.json(system, json.dumps(payload, default=str))
    except (LLMUnavailable, ValueError, AttributeError):
        return fallback


def grounded_or(text, evidence, fallback):
    """Accept LLM prose only if every number in it appears in the evidence."""
    if not isinstance(text, str) or not text.strip():
        return fallback
    return text if not ungrounded_numbers(text, [], evidence) else fallback
```

```python
    overview = f"{len(findings)} findings; most important: {findings[0]['summary']}" if findings else "No findings."
    res = narrate(llm, SYSTEM, [{"id": f["id"], "summary": f["summary"]} for f in findings], None)
    if isinstance(res, dict):
        order = [i for i in res.get("priority", []) if i in {f["id"] for f in findings}]
        if order:
            rank = {i: n for n, i in enumerate(order)}
            findings.sort(key=lambda f: rank.get(f["id"], len(rank)))
        if isinstance(res.get("overview"), str) and res["overview"].strip():
            overview = res["overview"]
    return {"findings": findings, "overview": overview}
```

```python
from __future__ import annotations

from typing import Any, TypedDict

from langgraph.graph import END, START, StateGraph

from .agents import advisor, analyzer, categorizer, forecaster, simulator, verifier
from .trace import trace as default_trace


class State(TypedDict, total=False):
    run_id: str
    profile: str
    request: dict
    df: Any
    opening: float
    corrections: dict
    goals: list
    findings: list
    overview: str
    forecast: dict
    model: Any
    balance: float
    recs: list
    dropped: list
    answer: dict


def route(state):
    return state["request"]["type"]


def _public(fc):
    return {k: v for k, v in fc.items() if k != "model"}


def build_graph(llm=None, tr=default_trace, persist=None):
    def emit(state, agent, kind, summary, data=None):
        tr.emit(state["run_id"], agent, kind, summary, data)

    def n_categorize(state):
        emit(state, "categorizer", "start", f"labeling {len(state['df'])} transactions")
        df = categorizer.categorize_df(state["df"], llm, state.get("corrections"))
        src = df["category_source"].value_counts().to_dict()
        emit(state, "categorizer", "result",
             f"{src.get('rule', 0)} by rule, {src.get('llm', 0)} by LLM, "
             f"{src.get('user', 0)} by your corrections, {src.get('none', 0)} uncategorized", src)
        return {"df": df}

    def n_analyze(state):
        emit(state, "analyzer", "tool", "running anomaly, subscription and behavior engines")
        out = analyzer.run(state["df"], llm)
        emit(state, "analyzer", "result", out["overview"], {"n": len(out["findings"])})
        return {"findings": out["findings"], "overview": out["overview"]}

    def n_forecast(state):
        emit(state, "forecaster", "tool", "fitting model and running 1,000 Monte Carlo paths")
        fc = forecaster.run(state["df"], state["opening"], state.get("goals") or [], llm)
        emit(state, "forecaster", "result", fc["narrative"], fc["summary"])
        return {"forecast": _public(fc), "model": fc["model"], "balance": fc["balance"]}

    def n_advise(state):
        cands = advisor.build_candidates(state["findings"], state["df"], state["forecast"],
                                         state["forecast"]["goals"])
        emit(state, "advisor", "tool", f"{len(cands)} candidate actions from findings and forecast")
        recs = advisor.advise(cands, llm, getattr(llm, "model", "") if llm else "")
        emit(state, "advisor", "result", f"drafted {len(recs)} recommendations")
        return {"recs": recs}

    def n_verify(state):
        emit(state, "verifier", "start", "recomputing every cited number from raw transactions")
        kept, dropped = verifier.verify_all(
            state["recs"], state["df"], llm,
            emit=lambda kind, summary, data=None: emit(state, "verifier", kind, summary, data))
        emit(state, "verifier", "result", f"{len(kept)} verified, {len(dropped)} dropped")
        return {"recs": kept, "dropped": dropped}

    def n_persist(state):
        if persist:
            persist(state)
        emit(state, "coordinator", "result", "saved results")
        return {}

    def n_prepare(state):
        if state.get("model") is None:
            fc = forecaster.run(state["df"], state["opening"], state.get("goals") or [], None)
            return {"model": fc["model"], "balance": fc["balance"], "forecast": _public(fc)}
        return {}

    def n_afford(state):
        emit(state, "simulator", "tool", "simulating the purchase against 1,000 futures")
        ans = simulator.afford(state["model"], state["balance"], state["request"], llm)
        emit(state, "simulator", "result", ans["narrative"])
        return {"answer": ans}

    def n_duel(state):
        emit(state, "simulator", "tool", "running both options on identical random futures")
        ans = simulator.run_duel(state["model"], state["balance"], state["request"], llm)
        emit(state, "simulator", "result", ans["reason"])
        return {"answer": ans}

    def n_chat(state):
        from .agents import chat
        emit(state, "chat", "tool", "answering from your data")
        ans = chat.answer(state["request"]["question"], state["df"], state["model"], state["balance"],
                          state.get("goals") or [], llm)
        emit(state, "chat", "result", ans["text"])
        return {"answer": ans}

    g = StateGraph(State)
    for name, fn in [("categorize", n_categorize), ("analyze", n_analyze), ("forecast", n_forecast),
                     ("advise", n_advise), ("verify", n_verify), ("persist", n_persist),
                     ("prepare", n_prepare), ("afford", n_afford), ("duel", n_duel), ("chat", n_chat)]:
        g.add_node(name, fn)
    g.add_conditional_edges(START, route, {"pipeline": "categorize", "afford": "prepare",
                                           "duel": "prepare", "chat": "prepare"})
    g.add_conditional_edges("prepare", route, {"afford": "afford", "duel": "duel", "chat": "chat"})
    for a, b in [("categorize", "analyze"), ("analyze", "forecast"), ("forecast", "advise"),
                 ("advise", "verify"), ("verify", "persist"), ("persist", END),
                 ("afford", END), ("duel", END), ("chat", END)]:
        g.add_edge(a, b)
    return g.compile()
```

```python
from __future__ import annotations

import io
import json
import os
import threading
import uuid

import pandas as pd

from . import db
from .engines.analyze import monthly_aggregates
from .graph import build_graph
from .llm import get_llm
from .trace import trace


def _clean(obj):
    return json.loads(json.dumps(obj, default=str))


def _sync():
    return os.getenv("FINPILOT_SYNC") == "1"


def parse_csv(raw: bytes) -> pd.DataFrame:
    try:
        df = pd.read_csv(io.BytesIO(raw))
    except Exception as e:
        raise ValueError(f"could not read CSV: {e}")
    df.columns = [c.strip().lower() for c in df.columns]
    missing = [c for c in ("date", "description", "amount") if c not in df.columns]
    if missing:
        raise ValueError(f"CSV must have columns date, description, amount (missing: {', '.join(missing)})")
    df = df[["date", "description", "amount"]].copy()
    df["date"] = pd.to_datetime(df["date"], errors="coerce")
    df["amount"] = pd.to_numeric(df["amount"], errors="coerce")
    df = df.dropna(subset=["date", "amount"])
    if len(df) < 5:
        raise ValueError("need at least 5 valid rows")
    return df.sort_values("date", kind="stable").reset_index(drop=True)


def _persist(profile):
    def save(state):
        with db.session() as s:
            db.save_transactions(s, profile, state["df"], state["opening"])
            db.save_findings(s, profile, _clean(state["findings"]))
            db.save_recs(s, profile, _clean(state["recs"]))
            db.save_forecast(s, profile, _clean({"forecast": state["forecast"], "overview": state["overview"]}))
            by_name = {g["name"]: g["pk"] for g in db.list_goals(s, profile)}
            for g in state["forecast"]["goals"]:
                if g["name"] in by_name:
                    db.append_goal_snapshot(s, by_name[g["name"]], _clean(
                        {"on": str(state["df"]["date"].max().date()), "p_met": g["p_met"],
                         "shortfall_p50": g["shortfall_p50"]}))
    return save


def _run(profile, raw_df, opening, run_id):
    try:
        with db.session() as s:
            corr, goals = db.corrections(s, profile), db.list_goals(s, profile)
        g = build_graph(get_llm(), trace, _persist(profile))
        g.invoke({"run_id": run_id, "profile": profile, "request": {"type": "pipeline"}, "df": raw_df,
                  "opening": float(opening), "corrections": corr, "goals": goals})
    except Exception as e:  # surface in the trace instead of dying silently
        trace.emit(run_id, "coordinator", "error", f"{type(e).__name__}: {e}")
    finally:
        trace.finish(run_id)
        with db.session() as s:
            db.save_trace(s, run_id)


def start_pipeline(profile, raw_df, opening):
    run_id = uuid.uuid4().hex[:10]
    trace.emit(run_id, "coordinator", "start", f"new data for '{profile}': {len(raw_df)} transactions")
    if _sync():
        _run(profile, raw_df, opening, run_id)
    else:
        threading.Thread(target=_run, args=(profile, raw_df, opening, run_id), daemon=True).start()
    return run_id


def load_persona(name):
    from eval.generate import load_persona as lp
    p = lp(name)
    return name, start_pipeline(name, p.df, p.opening_balance)


def autoseed(name="priya"):
    with db.session() as s:
        has = db.load_df(s, name)[0] is not None
    if not has:
        load_persona(name)


def request(profile, req):
    with db.session() as s:
        df, opening = db.load_df(s, profile)
        goals = db.list_goals(s, profile)
    if df is None:
        raise LookupError(f"no data for profile '{profile}'")
    if req["type"] in ("afford",) and not req.get("date"):
        req["date"] = str((df["date"].max() + pd.Timedelta(days=1)).date())
    if req["type"] == "duel":
        d0 = str((df["date"].max() + pd.Timedelta(days=1)).date())
        for k in ("a", "b"):
            req[k].setdefault("date", d0)
    run_id = uuid.uuid4().hex[:10]
    trace.emit(run_id, "coordinator", "start", f"{req['type']} request")
    out = build_graph(get_llm(), trace).invoke(
        {"run_id": run_id, "profile": profile, "request": req, "df": df, "opening": opening,
         "corrections": {}, "goals": goals})
    trace.finish(run_id)
    return run_id, _clean(out["answer"])


def state(profile):
    with db.session() as s:
        df, opening = db.load_df(s, profile)
        if df is None:
            return {"has_data": False}
        fc = db.load_forecast(s, profile) or {}
        out = {"has_data": True, "profile": profile, "findings": db.load_findings(s, profile),
               "recs": db.load_recs(s, profile), "goals": db.list_goals(s, profile),
               "forecast": fc.get("forecast"), "overview": fc.get("overview", ""),
               "balance": (fc.get("forecast") or {}).get("balance")}
    agg = monthly_aggregates(df).round(0)
    out["monthly"] = {"months": list(agg.index), "series": {c: agg[c].tolist() for c in agg.columns}}
    return _clean(out)
```

```python
def create_app():
    @asynccontextmanager
    async def lifespan(app):
        db.init()
        if os.getenv("FINPILOT_AUTOSEED") == "1":
            service.autoseed("priya")
        sched = None
        if os.getenv("FINPILOT_SCHEDULER") == "1":

            sched = build_scheduler()
            sched.start()
        yield
        if sched:
            sched.shutdown(wait=False)

    app = FastAPI(title="FinPilot", lifespan=lifespan)
    app.mount("/static", StaticFiles(directory=HERE / "static"), name="static")

    @app.get("/")
    def index():
        return FileResponse(HERE / "templates" / "index.html")

    @app.get("/api/personas")
    def personas():
        from eval.generate import PERSONAS
        return list(PERSONAS)

    @app.post("/api/load/{persona}")
    def load(persona: str):
        from eval.generate import PERSONAS
        if persona not in PERSONAS:
            raise HTTPException(404, "unknown persona")
        profile, run_id = service.load_persona(persona)
        return {"profile": profile, "run_id": run_id}

    @app.post("/api/upload")
    async def upload(profile: str = Form(...), opening_balance: float = Form(0.0), file: UploadFile = File(...)):
        try:
            df = service.parse_csv(await file.read())
        except ValueError as e:
            raise HTTPException(400, str(e))
        return {"profile": profile, "run_id": service.start_pipeline(profile, df, opening_balance)}

    @app.get("/api/state/{profile}")
    def state(profile: str):
        return service.state(profile)

    @app.get("/trace/{run_id}")
    def trace_json(run_id: str):
        return trace.since(run_id)

    @app.get("/trace/{run_id}/stream")
    async def trace_stream(run_id: str):
        async def gen():
            n = 0
            while True:
                for e in trace.since(run_id, n):
                    n += 1
                    yield {"event": "trace", "data": json.dumps(e, default=str)}
                if trace.is_done(run_id) and not trace.since(run_id, n):
                    break
                await asyncio.sleep(0.25)
        return EventSourceResponse(gen())

    def _ask(profile, req):
        try:
            run_id, answer = service.request(profile, req)
        except LookupError as e:
            raise HTTPException(404, str(e))
        return {"run_id": run_id, "answer": answer}

    @app.post("/api/afford/{profile}")
    def afford(profile: str, body: dict):
        return _ask(profile, {"type": "afford", **body})

    @app.post("/api/duel/{profile}")
    def duel(profile: str, body: dict):
        return _ask(profile, {"type": "duel", **body})

    @app.post("/api/chat/{profile}")
    def chat(profile: str, body: dict):
        return _ask(profile, {"type": "chat", "question": str(body.get("question", ""))})

    @app.get("/api/goals/{profile}")
    def goals_get(profile: str):
        with db.session() as s:
            return db.list_goals(s, profile)

    @app.post("/api/goals/{profile}")
    def goals_add(profile: str, body: dict):
        with db.session() as s:
            db.add_goal(s, profile, body["name"], body["target"], body["deadline"])
        return {"ok": True}

    @app.post("/api/correct/{profile}")
    def correct(profile: str, body: dict):
        from .config import CATEGORIES
        if body.get("category") not in CATEGORIES:
            raise HTTPException(400, "unknown category")
        with db.session() as s:
            db.add_correction(s, profile, body["merchant"].upper(), body["category"])
            df, opening = db.load_df(s, profile)
        if df is None:
            raise HTTPException(404, "no data")
        raw = df[["date", "description", "amount"]]
        return {"run_id": service.start_pipeline(profile, raw, opening)}

    return app


app = create_app()
```

```python
import pandas as pd

from .analyze import detect_recurring


def pace_alerts(df, today, min_over=0.3, min_abs=500.0):
    """Month-to-date spend per category vs the median of the same day-span in prior months."""
    today = pd.Timestamp(today).normalize()
    month = today.to_period("M")
    rec = {r["merchant"] for r in detect_recurring(df)}
    spend = df[(df["amount"] < 0) & (~df["merchant"].isin(rec))].copy()
    spend["m"] = spend["date"].dt.to_period("M")
    spend["day"] = spend["date"].dt.day
    spend["s"] = -spend["amount"]
    within = spend[spend["day"] <= today.day]
    out = []
    for cat, g in within.groupby("category"):
        by_m = g.groupby("m")["s"].sum()
        prior = by_m[by_m.index < month].tail(6)
        full_prior_months = sorted(m for m in spend["m"].unique() if m < month)
        if len(full_prior_months) < 3:
            continue
        prior = prior.reindex(full_prior_months[-6:], fill_value=0.0)
        mtd = float(by_m.get(month, 0.0))
        expected = float(prior.median())
        if mtd > expected * (1 + min_over) and mtd - expected >= min_abs:
            out.append({"id": f"pace:{cat}:{month}", "type": "pace_alert", "category": cat,
                        "mtd": round(mtd, 2), "expected": round(expected, 2),
                        "over_pct": round((mtd / expected - 1) * 100, 1) if expected else None,
                        "day": int(today.day)})
    return out
```

```python
def _tools(df, model, balance, goals):
    def month_total(category, month):
        m = df[(df["category"] == category) & (df["amount"] < 0) & (df["date"].dt.to_period("M").astype(str) == month)]
        return {"category": category, "month": month, "total": round(float(-m["amount"].sum()), 2), "n": int(len(m))}

    def top_merchants(n=5, month=None):
        d = df[df["amount"] < 0]
        if month:
            d = d[d["date"].dt.to_period("M").astype(str) == month]
        s = (-d.groupby("merchant")["amount"].sum()).sort_values(ascending=False).head(int(n))
        return {"month": month, "merchants": [{"merchant": k, "total": round(float(v), 2)} for k, v in s.items()]}

    def list_subscriptions():
        subs = detect_subscriptions(df)
        return {"subscriptions": [{"merchant": s["merchant"], "monthly_cost": s["monthly_cost"]} for s in subs],
                "total_monthly": round(sum(s["monthly_cost"] for s in subs), 2)}

    def forecast(horizon=30):
        return F.monte_carlo(model, balance, int(horizon)).summary() | {"horizon": int(horizon), "balance": round(balance, 2)}

    def afford(amount, date=None):
        on = date or str((model.last_date + pd.Timedelta(days=1)).date())
        r = S.simulate_purchase(model, balance, float(amount), on)
        return {"amount": float(amount), "date": on, "overdraft_risk_before": r["baseline"]["p_overdraft"],
                "overdraft_risk_after": r["with_purchase"]["p_overdraft"],
                "median_balance_after": r["with_purchase"]["end_p50"]}

    def goal_status():
        out = []
        for g in goals:
            out.append({"name": g["name"], "target": g["target"], "deadline": str(g["deadline"]),
                        **F.goal_status(model, balance, g["target"], pd.Timestamp(g["deadline"]))})
        return {"goals": out}

    return {"month_total": month_total, "top_merchants": top_merchants, "list_subscriptions": list_subscriptions,
            "forecast": forecast, "afford": afford, "goal_status": goal_status}


TOOL_SYSTEM = ("Choose a tool to answer the user's money question. Tools: month_total(category, month 'YYYY-MM'), "
               "top_merchants(n, month), list_subscriptions(), forecast(horizon), afford(amount, date), goal_status(). "
               "Categories: rent, groceries, dining, food_delivery, transport, utilities, subscriptions, shopping, "
               "entertainment, health, travel. Return JSON {\"tool\": name, \"args\": {...}}.")
ANSWER_SYSTEM = ("Answer the question using ONLY the tool results. Be concise, use full numbers like ₹1500. "
                 "Return JSON {\"text\": str}.")


def _heuristic(q):
    ql = q.lower()
    nums = [float(x.replace(",", "")) for x in re.findall(r"\d[\d,]*\.?\d*", q)]
    if "afford" in ql or "buy" in ql:
        return "afford", {"amount": nums[0] if nums else 0}
    if "subscri" in ql:
        return "list_subscriptions", {}
    if "goal" in ql:
        return "goal_status", {}
    if "top" in ql or "merchant" in ql or "where" in ql:
        return "top_merchants", {}
    return "forecast", {"horizon": 30}


def _fallback_text(name, res):
    if name == "list_subscriptions":
        subs = ", ".join(f"{s['merchant']} ({inr(s['monthly_cost'])})" for s in res["subscriptions"])
        return f"You pay for {len(res['subscriptions'])} subscriptions: {subs}. Total {inr(res['total_monthly'])}/month."
    if name == "afford":
        return (f"Buying {inr(res['amount'])} on {res['date']} moves your overdraft risk from "
                f"{res['overdraft_risk_before']*100:.0f}% to {res['overdraft_risk_after']*100:.0f}%.")
    if name == "month_total":
        return f"You spent {inr(res['total'])} on {res['category'].replace('_', ' ')} in {res['month']}."
    if name == "top_merchants":
        return "Top merchants: " + ", ".join(f"{m['merchant']} ({inr(m['total'])})" for m in res["merchants"]) + "."
    if name == "forecast":
        return (f"In {res['horizon']} days your median balance is {inr(res['end_p50'])} "
                f"with a {res['p_overdraft']*100:.0f}% overdraft risk.")
    return json.dumps(res)


def answer(question, df, model, balance, goals, llm=None):
    tools = _tools(df, model, balance, goals)
    name, args = None, {}
    if llm is not None:
        try:
            pick = llm.json(TOOL_SYSTEM, question)
            if pick.get("tool") in tools:
                name, args = pick["tool"], pick.get("args") or {}
        except (LLMUnavailable, ValueError, AttributeError):
            pass
    if name is None:
        name, args = _heuristic(question)
    try:
        result = tools[name](**args)
    except TypeError:
        result = tools[name]()
    fallback = _fallback_text(name, result)
    text = fallback
    if llm is not None:
        try:
            out = llm.json(ANSWER_SYSTEM, json.dumps({"question": question, "tool": name, "result": result}, default=str))
            text = grounded_or(out.get("text"), result, fallback)
        except (LLMUnavailable, ValueError, AttributeError):
            pass
    return {"text": text, "tool_calls": [{"name": name, "args": args, "result": result}]}
```

```python
def build_digest(profile):
    with db.session() as s:
        recs = [r for r in db.load_recs(s, profile) if r.get("verified")]
        fc = (db.load_forecast(s, profile) or {}).get("forecast")

    if not recs and not fc:
        return f"No data yet for {profile}."
    insight = fc["narrative"] if fc else "No forecast yet."
    best = max(recs, key=lambda r: r.get("est_monthly_savings", 0) + r.get("est_one_off", 0), default=None)
    lines = [f"FinPilot weekly digest for {profile}", "", f"Key insight: {insight}", ""]
    lines.append(f"One action: {best['text']}" if best else "One action: nothing urgent this week.")
    if best:
        lines.append(f"Why: triggered by {', '.join(best['finding_ids'])} (verified against your transactions).")
    for g in (fc or {}).get("goals", []):
        lines.append(f"Goal '{g['name']}': {g['p_met']*100:.0f}% chance of hitting {inr(g['target'])} by {g['deadline']}.")
    return "\n".join(lines)


def send_email(subject, body):
    host = os.getenv("SMTP_HOST")
    to = os.getenv("DIGEST_TO")
    if not host or not to:
        return False
    msg = EmailMessage()
    msg["Subject"], msg["To"], msg["From"] = subject, to, os.getenv("SMTP_USER", "finpilot@localhost")
    msg.set_content(body)
    try:
        with smtplib.SMTP(host, 587) as smtp:
            smtp.starttls()
            smtp.login(os.getenv("SMTP_USER", ""), os.getenv("SMTP_PASS", ""))
            smtp.send_message(msg)
        return True
    except Exception:
        return False


def deliver(profile):
    text = build_digest(profile)
    sent = send_email(f"FinPilot digest: {profile}", text)
    channel = "email" if sent else "ui"
    with db.session() as s:
        db.save_digest(s, profile, channel, text, sent)
    return {"channel": channel, "sent": sent, "content": text}
```

```python
def accuracy(y_true, y_pred):
    y_true, y_pred = list(y_true), list(y_pred)
    return sum(a == b for a, b in zip(y_true, y_pred)) / len(y_true)


def macro_f1(y_true, y_pred):
    y_true, y_pred = list(y_true), list(y_pred)
    f1s = []
    for c in sorted(set(y_true)):
        tp = sum(1 for t, p in zip(y_true, y_pred) if t == c and p == c)
        fp = sum(1 for t, p in zip(y_true, y_pred) if t != c and p == c)
        fn = sum(1 for t, p in zip(y_true, y_pred) if t == c and p != c)
        prec = tp / (tp + fp) if tp + fp else 0.0
        rec = tp / (tp + fn) if tp + fn else 0.0
        f1s.append(2 * prec * rec / (prec + rec) if prec + rec else 0.0)
    return sum(f1s) / len(f1s)


def precision_recall(pred, truth):
    pred, truth = set(pred), set(truth)
    tp = len(pred & truth)
    p = tp / len(pred) if pred else 1.0
    r = tp / len(truth) if truth else 1.0
    f = 2 * p * r / (p + r) if p + r else 0.0
    return {"precision": round(p, 3), "recall": round(r, 3), "f1": round(f, 3)}


def brier(probs, outcomes):
    return sum((p - o) ** 2 for p, o in zip(probs, outcomes)) / len(probs)
```

```python
        "large_charge": M.precision_recall({f["txn_id"] for f in A.txn_outliers(df, rec)}, t.anomaly_ids),
        "duplicate": M.precision_recall({f["txn_id"] for f in A.duplicate_charges(df)}, t.duplicate_ids),
        "category_spike": M.precision_recall({(f["month"], f["category"]) for f in A.category_spikes(df)}, {t.spike}),
        "subscriptions": M.precision_recall({s["merchant"] for s in subs},
                                            {"NETFLIX", "HOTSTAR", "SPOTIFY", "CULT FIT GYM"}),
        "unused_subscriptions": M.precision_recall({u["merchant"] for u in A.unused_subscriptions(df, subs)}, t.unused_subs),
        "payday_trigger": M.precision_recall({f["category"] for f in A.behavior_triggers(df, rec)},
                                             {t.payday_trigger_category}),
    }


def _holdout(df, opening, cutoff):
    train = df[df["date"] < cutoff]
    end = (cutoff + pd.offsets.MonthEnd(0)).normalize()
    test = df[(df["date"] >= cutoff) & (df["date"] <= end)]
    model = F.fit(train)
    bal0 = F.current_balance(train, opening)
    days = pd.date_range(cutoff, end)
    actual = bal0 + test.groupby(test["date"].dt.normalize())["amount"].sum().reindex(days, fill_value=0.0).cumsum()
    return train, model, bal0, days, actual.to_numpy()


def eval_forecast(persona, origins=3):
    df = _with_truth(persona)
    last_month = df["date"].max().to_period("M")
    acc = {"model": {"ape": [], "mape": [], "cov": []}, "naive": {"ape": [], "mape": [], "cov": []}}
    for k in range(origins):
        cutoff = (last_month - k).to_timestamp()
        train, model, bal0, days, actual = _holdout(df, persona.opening_balance, cutoff)
        model.last_date = cutoff - pd.Timedelta(days=1)  # forecast starts on the cutoff day
        r = F.monte_carlo(model, bal0, len(days), 1000)
        true_net = actual[-1] - bal0
        acc["model"]["ape"].append(abs((r.p50[-1] - bal0) - true_net) / max(abs(true_net), 1.0))
        acc["model"]["mape"].append(float(np.mean(np.abs(r.p50 - actual) / np.abs(actual))))
        acc["model"]["cov"].append(float(np.mean((actual >= r.p10) & (actual <= r.p90))))
        last30 = train[train["date"] >= cutoff - pd.Timedelta(days=30)]["amount"].sum() / 30
        naive = bal0 + last30 * np.arange(1, len(days) + 1)
        acc["naive"]["ape"].append(abs((naive[-1] - bal0) - true_net) / max(abs(true_net), 1.0))
        acc["naive"]["mape"].append(float(np.mean(np.abs(naive - actual) / np.abs(actual))))
    return {n: {"net_flow_ape": round(float(np.mean(v["ape"])), 4), "balance_mape": round(float(np.mean(v["mape"])), 4),
                "band_coverage": round(float(np.mean(v["cov"])), 4) if v["cov"] else None}
            for n, v in acc.items()}


def eval_calibration(personas, horizon=30, drop=0.10):
    """Event: balance falls more than `drop` below its starting level within `horizon` days."""
    probs, outcomes = [], []
    for p in personas:
        df = _with_truth(p)
        for cutoff in pd.date_range(df["date"].min() + pd.Timedelta(days=100),
                                    df["date"].max() - pd.Timedelta(days=horizon), freq="7D"):
            train = df[df["date"] < cutoff]
            model, bal0 = F.fit(train), F.current_balance(train, p.opening_balance)
            model.last_date = cutoff - pd.Timedelta(days=1)
            r = F.monte_carlo(model, bal0, horizon, 500)
            probs.append(float((r.paths.min(axis=1) < bal0 * (1 - drop)).mean()))
            test = df[(df["date"] >= cutoff) & (df["date"] < cutoff + pd.Timedelta(days=horizon))]
            days = pd.date_range(cutoff, periods=horizon)
            path = bal0 + test.groupby(test["date"].dt.normalize())["amount"].sum().reindex(days, fill_value=0.0).cumsum()
            outcomes.append(int(path.min() < bal0 * (1 - drop)))
    return {"n": len(probs), "mean_pred": round(float(np.mean(probs)), 4), "observed": round(float(np.mean(outcomes)), 4),
            "brier": round(M.brier(probs, outcomes), 4)}


def eval_grounding(persona, llm):
    tr = Trace()
    g = graph.build_graph(llm, tr)
    out = g.invoke({"run_id": "eval", "profile": "eval", "request": {"type": "pipeline"}, "df": persona.df.copy(),
                    "opening": persona.opening_balance, "corrections": {}, "goals": []})
    kinds = [e["kind"] for e in tr.since("eval") if e["agent"] == "verifier"]
    ok, caught = kinds.count("ok"), kinds.count("caught")
    return {"recs": ok + caught, "first_pass_ok": ok, "first_pass_caught": caught,
            "first_pass_grounding_rate": round(ok / max(ok + caught, 1), 4),
            "final_verified": sum(r["verified"] for r in out["recs"])}


def _avg(dicts):
    if not dicts or dicts[0] is None:
        return None
    if isinstance(dicts[0], dict):
        return {k: _avg([d[k] for d in dicts]) for k in dicts[0]}
    return round(float(np.mean(dicts)), 4)


def render_md(res):
    c, f = res["categorization"], res["forecast"]
    d = res["detection"]
    rows = ["# Evaluation results", "", f"Personas: {', '.join(res['personas'])} (synthetic, known ground truth)", "",
            "## Categorization", "| Method | Accuracy | Macro-F1 | Coverage |", "|---|---|---|---|"]
    for name in ("rules_only", "hybrid"):
        m = c.get(name)
        rows.append(f"| {name} | " + (" | ".join(str(m[k]) for k in ("accuracy", "macro_f1", "coverage")) if m else "n/a | n/a | n/a") + " |")
    rows += ["", "## Detection (precision / recall)", "| Task | Precision | Recall |", "|---|---|---|"]
    rows += [f"| {k} | {v['precision']} | {v['recall']} |" for k, v in d.items()]
    rows += ["", "## Forecast (held-out months)", "| Method | Net-flow APE | Balance MAPE | p10-p90 coverage |", "|---|---|---|---|"]
    for n in ("model", "naive"):
        rows.append(f"| {n} | {f[n]['net_flow_ape']} | {f[n]['balance_mape']} | {f[n]['band_coverage'] or '-'} |")
    cal = res["calibration"]
    rows += ["", f"## Calibration\nn={cal['n']}, mean predicted {cal['mean_pred']}, observed {cal['observed']}, Brier {cal['brier']}", ""]
    g = res.get("grounding")
    rows.append("## Grounding\n" + (f"first-pass grounding rate {g['first_pass_grounding_rate']} "
                                    f"({g['first_pass_caught']} caught of {g['recs']}), final verified {g['final_verified']}" if g else "n/a (--no-llm)"))
    return "\n".join(rows)


def main(argv=None):
    argv = argv if argv is not None else sys.argv[1:]
    use_llm = "--no-llm" not in argv
    llm = LLM() if use_llm else None
    personas = {n: load_persona(n) for n in PERSONAS}
    res = {"personas": list(personas), "model": getattr(llm, "model", None)}
    cats = [eval_categorization(p, llm) for p in personas.values()]
    res["categorization"] = {"rules_only": _avg([c["rules_only"] for c in cats]),
                             "hybrid": _avg([c["hybrid"] for c in cats]) if use_llm else None}
    res["detection"] = _avg([eval_detection(p) for p in personas.values()])
    res["forecast"] = _avg([eval_forecast(p) for p in personas.values()])
    res["calibration"] = eval_calibration(list(personas.values()))
    res["grounding"] = eval_grounding(personas["priya"], llm) if use_llm else None
    (OUT / "results.json").write_text(json.dumps(res, indent=2))
    (OUT / "results.md").write_text(render_md(res))
    print(render_md(res))
    return res


if __name__ == "__main__":
    main()
```

```python
"""Advice backtest: apply the system at a past cutoff, then measure what following its advice
would have saved over the following period. This is an upper bound that assumes the user
complied; it is NOT a prediction of behaviour."""
from __future__ import annotations

import pandas as pd

from eval.run_eval import _with_truth
from finpilot.agents.advisor import build_candidates
from finpilot.engines.analyze import analyze_all


def run(persona, days=90):
    df = _with_truth(persona)
    cutoff = df["date"].max().normalize() - pd.Timedelta(days=days)
    train, test = df[df["date"] <= cutoff], df[df["date"] > cutoff]
    cands = build_candidates(analyze_all(train), train)
    followed, savings = [], 0.0
    cancelled = set()
    for c in cands:
        e = c["evidence"]
        if c["trigger_type"] == "unused_subscription":
            spent = float(-test[(test["merchant"] == e["merchant"]) & (test["amount"] < 0)]["amount"].sum())
            savings += spent
            cancelled.add(e["merchant"])
            followed.append({"action": c["action"], "saved": round(spent, 2)})
        elif c["trigger_type"] == "category_spike":
            t = test[(test["category"] == e["category"]) & (test["amount"] < 0)]
            monthly = (-t.groupby(t["date"].dt.to_period("M"))["amount"].sum())
            excess = float((monthly - e["baseline"]).clip(lower=0).sum())
            savings += excess
            followed.append({"action": c["action"], "saved": round(excess, 2)})
    total = float(-test[test["amount"] < 0]["amount"].sum())
    precision = (len(cancelled & persona.truth.unused_subs) / len(cancelled)) if cancelled else 1.0
    return {"cutoff": str(cutoff.date()), "followed": followed, "potential_savings": round(savings, 2),
            "pct_of_spend": round(savings / total, 4) if total else 0.0, "cancel_precision": round(precision, 3)}
```

```python
"""LLM-only baselines. They receive the same raw data summary but no engines, to test whether
the hybrid design is worth its complexity."""
from __future__ import annotations

import json

import numpy as np
import pandas as pd

from eval import metrics as M
from eval.run_eval import _with_truth, eval_categorization
from finpilot.agents.categorizer import _llm_map
from finpilot.agents.verifier import audit
from finpilot.engines.categorize import normalize_merchant
from finpilot.llm import LLMUnavailable


def llm_only_categorization(persona, llm):
    merchants = sorted({normalize_merchant(d) for d in persona.df["description"]})
    mapping = _llm_map(set(merchants), llm)
    pred = [mapping.get(normalize_merchant(d), "uncategorized") for d in persona.df["description"]]
    truth = persona.truth.categories
    return {"accuracy": round(M.accuracy(truth, pred), 4), "macro_f1": round(M.macro_f1(truth, pred), 4)}


ADVICE_SYSTEM = ("You are a financial advisor. From the monthly spend by merchant, write up to 6 specific recommendations. "
                 "Return JSON {\"recommendations\":[{\"text\":str,\"claims\":[{\"kind\":\"monthly_cost\"|\"annual_cost\"|\"month_total\"|\"pct_change\","
                 "\"merchant\":str (for cost kinds),\"category\":str,\"month\":\"YYYY-MM\" (for month kinds),\"value\":number}]}]}")


def _merchant_table(df):
    s = df[df["amount"] < 0].copy()
    s["month"] = s["date"].dt.to_period("M").astype(str)
    t = (-s.groupby(["merchant", "month"])["amount"].sum()).round(0).unstack(fill_value=0)
    return t.to_csv()


def llm_only_advice(persona, llm):
    df = _with_truth(persona)
    try:
        res = llm.json(ADVICE_SYSTEM, _merchant_table(df))
    except (LLMUnavailable, ValueError):
        return {"recs": 0, "with_issues": 0, "hallucination_rate": None, "ungrounded_numbers": 0}
    recs = [r for r in res.get("recommendations", []) if "text" in r]
    with_issues, ungrounded = 0, 0
    for r in recs:
        rec = {"text": r["text"], "claims": r.get("claims") or [], "evidence": {}}
        issues = audit(rec, df)
        with_issues += bool(issues)
        ungrounded += sum(i["type"] == "ungrounded_number" for i in issues)
    return {"recs": len(recs), "with_issues": with_issues,
            "hallucination_rate": round(with_issues / len(recs), 4) if recs else None,
            "ungrounded_numbers": ungrounded}


def llm_only_forecast(persona, llm):
    df = _with_truth(persona)
    month = df["date"].max().to_period("M")
    train, test = df[df["date"].dt.to_period("M") < month], df[df["date"].dt.to_period("M") == month]
    hist = train.groupby(train["date"].dt.to_period("M").astype(str))["amount"].sum().round(0)
    try:
        res = llm.json("Given monthly net cash flow history, predict next month's net flow. "
                       "Return JSON {\"next_month_net_flow\": number}.", hist.to_csv())
        pred = float(res["next_month_net_flow"])
    except (LLMUnavailable, ValueError, KeyError, TypeError):
        return {"net_flow_ape": None}
    true = float(test["amount"].sum())
    return {"net_flow_ape": round(abs(pred - true) / max(abs(true), 1.0), 4)}


def run(personas, llm):
    out = {"categorization": {"rules_only": None, "llm_only": None, "hybrid": None},
           "advice": {"llm_only": None}, "forecast": {"llm_only": None}}
    ps = list(personas.values())
    cats = [eval_categorization(p, llm) for p in ps]
    out["categorization"]["rules_only"] = {k: round(float(np.mean([c["rules_only"][k] for c in cats])), 4)
                                           for k in ("accuracy", "macro_f1")}
    out["categorization"]["hybrid"] = {k: round(float(np.mean([c["hybrid"][k] for c in cats])), 4)
                                       for k in ("accuracy", "macro_f1")}
    lo = [llm_only_categorization(p, llm) for p in ps]
    out["categorization"]["llm_only"] = {k: round(float(np.mean([x[k] for x in lo])), 4) for k in ("accuracy", "macro_f1")}
    adv = [llm_only_advice(p, llm) for p in ps]
    recs = sum(a["recs"] for a in adv)
    out["advice"]["llm_only"] = {"recs": recs, "with_issues": sum(a["with_issues"] for a in adv),
                                 "hallucination_rate": round(sum(a["with_issues"] for a in adv) / recs, 4) if recs else None}
    fa = [llm_only_forecast(p, llm)["net_flow_ape"] for p in ps]
    fa = [x for x in fa if x is not None]
    out["forecast"]["llm_only"] = {"net_flow_ape": round(float(np.mean(fa)), 4) if fa else None}
    return out
```

```python
"""Run categorization and detection on a public transaction CSV.

Usage: python -m eval.real_data path.csv eval/real_map.json
real_map.json: {"date": "<col>", "description": "<col>", "amount": "<col>",
                "label": "<col or null>", "label_map": {"<dataset label>": "<our category>"},
                "negate_amount": false, "source": "<dataset name + licence>"}
"""
from __future__ import annotations

import json
import sys

import pandas as pd

from eval import metrics as M
from finpilot.agents.categorizer import categorize_df
from finpilot.engines import analyze as A
from finpilot.llm import LLM


def run(csv_path, mapping_path, llm):
    m = json.load(open(mapping_path))
    raw = pd.read_csv(csv_path)
    df = pd.DataFrame({"date": pd.to_datetime(raw[m["date"]], errors="coerce"),
                       "description": raw[m["description"]].astype(str),
                       "amount": pd.to_numeric(raw[m["amount"]], errors="coerce")})
    if m.get("negate_amount"):
        df["amount"] = -df["amount"]
    keep = df.dropna().index
    df = df.loc[keep].sort_values("date").reset_index(drop=True)
    labels = raw.loc[keep, m["label"]].reset_index(drop=True) if m.get("label") else None
    if labels is not None:
        labels = labels.reindex(range(len(raw.loc[keep]))).reset_index(drop=True)
    cat = categorize_df(df, llm)
    out = {"source": m.get("source"), "rows": len(df),
           "source_mix": cat["category_source"].value_counts().to_dict()}
    if labels is not None:
        truth = labels.map(m["label_map"])
        mask = truth.notna()
        out["labelled_rows"] = int(mask.sum())
        out["accuracy_vs_dataset_labels"] = round(M.accuracy(truth[mask], cat["category"][mask]), 4)
        out["macro_f1_vs_dataset_labels"] = round(M.macro_f1(truth[mask], cat["category"][mask]), 4)
    findings = A.analyze_all(cat)
    out["findings_by_type"] = pd.Series([f["type"] for f in findings]).value_counts().to_dict()
    out["sample_for_manual_review"] = [f.get("merchant") or f.get("category") for f in findings[:10]]
    return out


if __name__ == "__main__":
    print(json.dumps(run(sys.argv[1], sys.argv[2], LLM()), indent=2, default=str))
```
